--- 6.1 Create & Backfill buyer_accounts, artisan_accounts, admin_accounts ---
USE `hunarlink`;

CREATE TABLE IF NOT EXISTS `buyer_accounts` (
  `id` CHAR(36) CHARACTER SET ascii COLLATE ascii_bin NOT NULL,
  `clerk_user_id` VARCHAR(128) NOT NULL,
  `email` VARCHAR(255) NOT NULL,
  `email_verified` TINYINT(1) NOT NULL DEFAULT 0,
  `phone` VARCHAR(32) NULL,
  `first_name` VARCHAR(100) NOT NULL,
  `last_name` VARCHAR(100) NULL,
  `display_name` VARCHAR(160) NULL,
  `preferred_currency` CHAR(3) NOT NULL DEFAULT 'INR',
  `account_status` VARCHAR(32) NOT NULL DEFAULT 'ACTIVE',
  `last_login_at` DATETIME(6) NULL,
  `created_at` DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  `updated_at` DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_buyer_accounts_clerk_user_id` (`clerk_user_id`),
  UNIQUE KEY `uq_buyer_accounts_email` (`email`),
  CONSTRAINT `fk_buyer_accounts_user` FOREIGN KEY (`id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE IF NOT EXISTS `artisan_accounts` (
  `id` CHAR(36) CHARACTER SET ascii COLLATE ascii_bin NOT NULL,
  `clerk_user_id` VARCHAR(128) NOT NULL,
  `email` VARCHAR(255) NOT NULL,
  `email_verified` TINYINT(1) NOT NULL DEFAULT 0,
  `phone` VARCHAR(32) NULL,
  `first_name` VARCHAR(100) NOT NULL,
  `last_name` VARCHAR(100) NULL,
  `display_name` VARCHAR(160) NULL,
  `store_name` VARCHAR(160) NULL,
  `craft_specialty` VARCHAR(160) NULL,
  `workshop_district` VARCHAR(100) NULL,
  `verification_status` VARCHAR(32) NOT NULL DEFAULT 'UNVERIFIED',
  `account_status` VARCHAR(32) NOT NULL DEFAULT 'ACTIVE',
  `last_login_at` DATETIME(6) NULL,
  `created_at` DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  `updated_at` DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_artisan_accounts_clerk_user_id` (`clerk_user_id`),
  UNIQUE KEY `uq_artisan_accounts_email` (`email`),
  CONSTRAINT `fk_artisan_accounts_user` FOREIGN KEY (`id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE IF NOT EXISTS `admin_accounts` (
  `id` CHAR(36) CHARACTER SET ascii COLLATE ascii_bin NOT NULL,
  `email` VARCHAR(255) NOT NULL,
  `password_hash` VARCHAR(255) NOT NULL,
  `first_name` VARCHAR(100) NOT NULL,
  `last_name` VARCHAR(100) NULL,
  `display_name` VARCHAR(160) NULL,
  `department` VARCHAR(100) NULL DEFAULT 'Platform Administration',
  `account_status` VARCHAR(32) NOT NULL DEFAULT 'ACTIVE',
  `is_active` TINYINT(1) NOT NULL DEFAULT 1,
  `last_login_at` DATETIME(6) NULL,
  `created_at` DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  `updated_at` DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_admin_accounts_email` (`email`),
  CONSTRAINT `fk_admin_accounts_user` FOREIGN KEY (`id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

INSERT IGNORE INTO `admin_accounts` (
  `id`, `email`, `password_hash`, `first_name`, `last_name`, `display_name`,
  `department`, `account_status`, `is_active`, `last_login_at`, `created_at`, `updated_at`
)
SELECT
  u.`id`, u.`email`, COALESCE(ac.`password_hash`, 'REVOKED'), u.`first_name`, u.`last_name`,
  u.`display_name`, COALESCE(ap.`department`, 'Platform Administration'), u.`account_status`,
  COALESCE(ac.`is_active`, 1), COALESCE(ac.`last_login_at`, u.`last_login_at`), u.`created_at`, u.`updated_at`
FROM `users` u
LEFT JOIN `admin_credentials` ac ON ac.`user_id` = u.`id`
LEFT JOIN `admin_profiles` ap ON ap.`user_id` = u.`id`
WHERE ac.`id` IS NOT NULL OR ap.`id` IS NOT NULL;

INSERT IGNORE INTO `artisan_accounts` (
  `id`, `clerk_user_id`, `email`, `email_verified`, `phone`, `first_name`, `last_name`,
  `display_name`, `store_name`, `craft_specialty`, `workshop_district`,
  `verification_status`, `account_status`, `last_login_at`, `created_at`, `updated_at`
)
SELECT
  u.`id`, u.`clerk_user_id`, u.`email`, u.`email_verified`, u.`phone`, u.`first_name`, u.`last_name`,
  u.`display_name`, sp.`store_name`, sp.`craft_specialty`, sp.`workshop_district`,
  COALESCE(sp.`verification_status`, 'UNVERIFIED'), u.`account_status`, u.`last_login_at`, u.`created_at`, u.`updated_at`
FROM `users` u
JOIN `seller_profiles` sp ON sp.`user_id` = u.`id`
LEFT JOIN `admin_accounts` aa ON aa.`id` = u.`id`
WHERE aa.`id` IS NULL;

INSERT IGNORE INTO `buyer_accounts` (
  `id`, `clerk_user_id`, `email`, `email_verified`, `phone`, `first_name`, `last_name`,
  `display_name`, `preferred_currency`, `account_status`, `last_login_at`, `created_at`, `updated_at`
)
SELECT
  u.`id`, u.`clerk_user_id`, u.`email`, u.`email_verified`, u.`phone`, u.`first_name`, u.`last_name`,
  u.`display_name`, u.`preferred_currency`, u.`account_status`, u.`last_login_at`, u.`created_at`, u.`updated_at`
FROM `users` u
LEFT JOIN `admin_accounts` aa ON aa.`id` = u.`id`
LEFT JOIN `artisan_accounts` ara ON ara.`id` = u.`id`
WHERE aa.`id` IS NULL AND ara.`id` IS NULL;

--- 6.2 Useful Live Inspection Queries in MySQL Workbench ---
USE `hunarlink`;

-- Count accounts in each dedicated table:
SELECT 'buyer_accounts' AS table_name, COUNT(*) AS rows_count FROM `buyer_accounts`
UNION ALL
SELECT 'artisan_accounts', COUNT(*) FROM `artisan_accounts`
UNION ALL
SELECT 'admin_accounts', COUNT(*) FROM `admin_accounts`;

-- View newest Buyers, Artisans, and Admins:
SELECT * FROM `buyer_accounts` ORDER BY `created_at` DESC LIMIT 25;
SELECT * FROM `artisan_accounts` ORDER BY `created_at` DESC LIMIT 25;
SELECT `id`, `email`, `display_name`, `department`, `is_active`, `last_login_at` FROM `admin_accounts`;

-- View newest Retail Orders & Razorpay Payments:
SELECT `order_number`, `order_status`, `payment_status`, `grand_total_amount`, `created_at`
FROM `retail_orders` ORDER BY `created_at` DESC LIMIT 20;

SELECT `payment_reference`, `gateway_order_id`, `gateway_payment_id`, `amount`, `payment_status`, `captured_at`
FROM `payments` ORDER BY `created_at` DESC LIMIT 20;

-- View ASIGS Guardrail Cases & Immutable Audit Logs:
SELECT `case_number`, `entity_type`, `risk_level`, `case_status`, `created_at`
FROM `guardrail_cases` ORDER BY `created_at` DESC LIMIT 20;

SELECT `action`, `actor_type`, `entity_type`, `created_at`
FROM `audit_logs` ORDER BY `created_at` DESC LIMIT 25;

7. REDIS COMMANDS (WINDOWS LOCAL CACHE & RATE LIMITER)
================================================================================
Note: FastAPI runs even if Redis is offline (automatic 15s circuit-breaker
fallback to MySQL). To run live Redis locally on Windows:

# Option A: Docker Desktop (Recommended 1-liner)
docker run -d --name hunarlink-redis --restart unless-stopped -p 127.0.0.1:6379:6379 redis:7.4-alpine

# Check Redis container & ping:
docker exec -it hunarlink-redis redis-cli ping
# Expected output: PONG

# Inspect live HunarLink Redis keys (dev only):
docker exec -it hunarlink-redis redis-cli --scan --pattern "hl:*"

# Clear only HunarLink cache keys if needed during manual DB edits:
docker exec -it hunarlink-redis redis-cli FLUSHDB
