# ============================================================
# 5. MySQL — Create DB + app user (run as root)
# ============================================================
mysql -u root -p -e "SELECT VERSION();"
mysql -u root -p -e "CREATE DATABASE IF NOT EXISTS hunarlink CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci;"
mysql -u root -p -e "CREATE USER IF NOT EXISTS 'hunarlink_app'@'127.0.0.1' IDENTIFIED WITH caching_sha2_password BY 'YOUR_STRONG_PASSWORD';"
mysql -u root -p -e "GRANT SELECT, INSERT, UPDATE, DELETE, EXECUTE ON hunarlink.* TO 'hunarlink_app'@'127.0.0.1'; FLUSH PRIVILEGES;"

# --- Alternative: Single-file installer ---
cd hunarlink-database
Get-Content hunarlink_mysql_full_install.sql | mysql -u root -p
